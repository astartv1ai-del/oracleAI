  quality:
    runs-on: ubuntu-latest

          sudo apt-get update
          sudo apt-get install -y --no-install-recommends \
            libcairo2 libpango-1.0-0 libgdk-pixbuf-2.0-0 libffi-dev sqlite3

      - name: Зависимости (закреплённые версии)
        run: python -m pip install -r requirements-dev.txt

      - name: Ruff
        run: python -m ruff check app scripts tests

      - name: Python syntax/import smoke
        run: python -m compileall -q app scripts tests

      - name: JavaScript syntax
        shell: bash
        run: |
          find miniapp/js admin -type f -name '*.js' -print0 \
            | xargs -0 -r -n1 node --check

      - name: Cache-busting policy
        run: python scripts/check_cache_busting.py

      - name: Design contract
        run: python scripts/check_design_contract.py

      - name: LLM golden-set evaluator smoke
        run: |
          mkdir -p artifacts
          python scripts/generate_eval_set.py
          python scripts/evaluate_llm.py \
            --cases data/llm_eval/golden_cases.jsonl \
            --responses data/llm_eval/sample_responses.jsonl \
            --out artifacts/llm-eval-report.json \
            --min-score 0.75

      - name: Migration-focused tests
        run: python -m pytest tests/test_migrations.py tests/test_data.py -q -p no:cacheprovider --timeout=120

      - name: Full tests
        run: python -m pytest tests/ -q -p no:cacheprovider --timeout=120

      - name: Dependency audit
        run: python -m pip_audit -r requirements.txt

      - name: Selfcheck
        run: python scripts/selfcheck.py

      - name: Production-readiness static gate
        run: python scripts/release_gate.py

      - name: Upload release evidence
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: release-evidence-${{ github.sha }}
          path: artifacts/llm-eval-report.json
          if-no-files-found: ignore
