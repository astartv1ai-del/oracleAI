# Запуск на VPS:  docker compose -f infra/docker-compose.yml up -d --build
# Домен в Caddyfile + WEBAPP_URL=https://домен в ../.env
#
# Процессов ровно два, и это принципиально: планировщик и рассылки живут ТОЛЬКО
# в процессе бота. Если поднять второй бот, клиентки получат каждое сообщение
# дважды. API масштабируется свободно — он ничего не рассылает.

services:
  bot:
    build: { context: .., dockerfile: infra/Dockerfile }
    env_file: ../.env
    volumes: [ oracle_data:/opt/oracle/data ]
    restart: unless-stopped
    stop_grace_period: 30s        # успеть закрыть WAL и сохранить прогресс рассылки
    healthcheck:
      # HTTP бот не поднимает, поэтому liveness решается по хартбиту планировщика
      # (см. scripts/healthcheck.py): бот жив, пока тик регулярно обновляет метку
      test: ["CMD", "python", "/opt/oracle/scripts/healthcheck.py"]
      interval: 60s
      timeout: 10s
      retries: 3
      start_period: 120s
    logging:
      driver: json-file
      options: { max-size: "10m", max-file: "5" }

  api:
    build: { context: .., dockerfile: infra/Dockerfile }
    # 1 воркер обязателен: rate-limiter `_hits` и бакет исходящих — in-process
    # (G22). Несколько воркеров дадут каждому свой запас и удвоят бюджет.
    command: >
      uvicorn app.api.main:app --host 0.0.0.0 --port 8080 --workers 1
    env_file: ../.env
    volumes: [ oracle_data:/opt/oracle/data ]
    restart: unless-stopped
    expose: [ "8080" ]
    healthcheck:
      test: ["CMD", "curl", "-fsS", "http://localhost:8080/api/health"]
      interval: 30s
      timeout: 5s
      retries: 3
      start_period: 20s
    logging:
      driver: json-file
      options: { max-size: "10m", max-file: "5" }

  caddy:
    image: caddy:2
    env_file: ../.env
    ports: [ "80:80", "443:443" ]
    volumes:
      - ./Caddyfile:/etc/caddy/Caddyfile:ro
      - caddy_data:/data
      - caddy_config:/config
    restart: unless-stopped
    depends_on:
      api:
        condition: service_healthy

  # Ежедневный backup чувствительной SQLite-базы. Production требует ключ:
  # raw .db snapshots не создаются, копия получает checksum и уходит off-site.
  backup:
    image: alpine:3
    volumes:
      - oracle_data:/data:ro
      - oracle_backups:/backups
      - ${BACKUP_ENCRYPTION_KEY_HOST_PATH:-/etc/oracle/backup.key}:/run/secrets/oracle-backup-key:ro
    environment:
      BACKUP_S3_URL: ${BACKUP_S3_URL:-}
      BACKUP_S3_ACCESS_KEY: ${BACKUP_S3_ACCESS_KEY:-}
      BACKUP_S3_SECRET_KEY: ${BACKUP_S3_SECRET_KEY:-}
      BACKUP_S3_BUCKET: ${BACKUP_S3_BUCKET:-}
      BACKUP_S3_REGION: ${BACKUP_S3_REGION:-us-east-1}
      BACKUP_ENCRYPTION_KEY_FILE: /run/secrets/oracle-backup-key
      BACKUP_REQUIRE_ENCRYPTION: ${BACKUP_REQUIRE_ENCRYPTION:-1}
      BACKUP_KEEP: ${BACKUP_KEEP:-14}
    entrypoint: >
      sh -c "apk add --no-cache openssl sqlite s3cmd >/dev/null &&
             [ -r \"\$BACKUP_ENCRYPTION_KEY_FILE\" ] ||
               { echo 'backup key is required'; exit 1; };
             while true; do
               ts=\$$(date -u +%Y%m%d-%H%M%S);
               tmp=/backups/.oracle-\$\$ts.db.tmp;
               file=/backups/oracle-\$\$ts.db.enc;
               sqlite3 /data/oracle.db \".backup '\$\$tmp'\" &&
               [ \"\$(sqlite3 \"\$\$tmp\" 'PRAGMA integrity_check;')\" = ok ] &&
               openssl enc -aes-256-cbc -pbkdf2 -iter 200000 -salt \
                 -pass file:\"\$BACKUP_ENCRYPTION_KEY_FILE\" \
                 -in \"\$\$tmp\" -out \"\$\$file\" &&
               rm -f \"\$\$tmp\" &&
               sha256sum \"\$\$file\" > \"\$\$file.sha256\" &&
               echo \"encrypted backup ok: `basename \"\$\$file\"`\";
               if [ -n \"\$BACKUP_S3_URL\" ] && [ -n \"\$BACKUP_S3_BUCKET\" ]; then
                 export AWS_ACCESS_KEY_ID=\$BACKUP_S3_ACCESS_KEY AWS_SECRET_ACCESS_KEY=\$BACKUP_S3_SECRET_KEY;
                 s3cmd --host=\$BACKUP_S3_URL --region=\$BACKUP_S3_REGION \
                       put \"\$\$file\" s3://\$BACKUP_S3_BUCKET/ &&
                 s3cmd --host=\$BACKUP_S3_URL --region=\$BACKUP_S3_REGION \
                       put \"\$\$file.sha256\" s3://\$BACKUP_S3_BUCKET/;
               fi;
               find /backups -maxdepth 1 -type f -name 'oracle-*.db.enc' -mtime +14 -delete;
               find /backups -maxdepth 1 -type f -name 'oracle-*.db.enc.sha256' -mtime +14 -delete;
               sleep 86400;
             done"
    restart: unless-stopped

volumes:
  oracle_data:
  oracle_backups:
  caddy_data:
  caddy_config:

# Один образ на оба процесса (бот и API) — они делят код и базу.
# Команда по умолчанию — бот; API поднимается тем же образом с своим CMD
# (см. infra/docker-compose.yml).
FROM python:3.12-slim

# Системные библиотеки: cairo/pango нужны WeasyPrint (PDF-разборы),
# libjpeg/zlib — Pillow (карточки для сторис). tzdata нужен zoneinfo:
# без него планировщик не сможет посчитать «9 утра по её таймзоне».
RUN apt-get update && apt-get install -y --no-install-recommends \
        tzdata \
        libcairo2 libpango-1.0-0 libpangocairo-1.0-0 libgdk-pixbuf-2.0-0 \
        libffi8 shared-mime-info \
        libjpeg62-turbo zlib1g \
        fonts-dejavu-core \
        curl \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /opt/oracle

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY app ./app
COPY miniapp ./miniapp
COPY admin ./admin
COPY scripts ./scripts

# Пишем только в data/ — остальное дерево может быть read-only
RUN useradd --create-home --uid 10001 oracle \
    && mkdir -p /opt/oracle/data \
    && chown -R oracle:oracle /opt/oracle
USER oracle

ENV PYTHONUNBUFFERED=1 \
    PYTHONDONTWRITEBYTECODE=1 \
    DATA_DIR=/opt/oracle/data

CMD ["python", "-m", "app.bot.main"]
